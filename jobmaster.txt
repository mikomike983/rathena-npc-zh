// 檢查玩家是否達到所需等級.
// 若未達到則關閉, 若達到則返回
function	Require_Level	{
	if (BaseLevel < getarg(0) || JobLevel < getarg(1)) {
		.@blvl = getarg(0) - BaseLevel;
		.@jlvl = getarg(1) - JobLevel;
		mes "等級需求:";
		mes ((getarg(0)>1)? 
			"^bb0000"+getarg(0)+"^000000 (^bb0000基本等級^000000) / ":"")+"^00bb00"+
			getarg(1)+"^000000 (^00bb00職業等級^000000)";
		mes "你需要再提升 " +
			((.@blvl > 0) ? "^bb0000"+.@blvl+"^000000 級基本等級 " + 
				((.@jlvl > 0) ? "以及 " : "") : "") +
			((.@jlvl > 0) ? "^00bb00"+.@jlvl+"^000000 級職業等級 " : "") +
			"才能繼續.";
		close;
	}
	return;
}

// 檢查指定的職業代碼是否為寶寶職業
function Is_Baby	{
	return ((getarg(0, eaclass())&EAJL_BABY)>0);
}

// 檢查玩家是否可以轉職為四轉職業.
// 注意: 此處不包含等級檢查.
function	Can_Change_Fourth	{
	if( !(eaclass()&EAJL_THIRD) )
		return false;
	if( Class >= Job_Dragon_Knight && Class <= Job_Spirit_Handler )
		return false;
	return true;
}

// 檢查玩家是否可以轉職為三轉職業.
// 注意: 此處不包含等級檢查.
function	Can_Change_Third	{
	// 若要轉職為三轉職業你必須是以下之一:
	// * 二轉職業
	// * 進階二轉職業
	// * 寶寶二轉職業
	if( !.ThirdClass )
		return false; // 已停用三轉轉職
	if( !(eaclass()&EAJL_2) )
		return false; // 非二轉職業
	if( eaclass()&EAJL_THIRD )
		return false; // 已經是三轉職業
	if( roclass(eaclass()|EAJL_THIRD) < 0 )
		return false; // 該職業沒有三轉職業
	if( (eaclass()&EAJ_UPPERMASK) == EAJ_SUPER_NOVICE )
		return false; // 擴充超級初學者等同三轉, 但有獨立的判斷分支
	if( Is_Baby() && (!.BabyClass || !.BabyThird) )
		return false; // 不允許寶寶 (三轉) 轉職
	return true;
}

function	Can_Rebirth	{
	// 若要轉生, 你必須是:
	// * 二轉職業
	if( !.RebirthClass )
		return false; // 已停用轉生
	if( !(eaclass()&EAJL_2) )
		return false; // 非二轉職業
	if( eaclass()&(EAJL_UPPER|EAJL_THIRD) )
		return false; // 已經轉生/ 三轉職業
	if( roclass(eaclass()|EAJL_UPPER) < 0 )
		return false; // 該職業沒有進階職業
	if( Is_Baby() && !.BabyClass )
		return false; // 不允許寶寶職業轉職
	return true;
}

// 檢查指定的職業代碼是否為一轉職業
function	Is_First_Cls	{

}

function	Check_Riding	{
	// 注意: 為什麼我們總是需要檢查騎乘狀態:
	// 坐騎會被視為另一種職業, 這會讓這個 NPC
	// 為了處理這些特殊情況而變得更龐大.
	if (checkfalcon() || checkcart() || checkriding() || ismounting()) {
		mes "在繼續之前請先解除你的" +
			((checkfalcon()) ? "獵鷹" : "") +
			((checkcart()) ? "手推車" : "") +
			((checkriding()) ? "大嘴鳥" : "") +
			((ismounting()) ? "坐騎" : "") +
			".";
		close;
	}
	return;
}
function	Check_SkillPoints	{
	if (.SkillPointCheck && SkillPoint) {
		mes "在繼續之前請先將所有技能點數使用完畢.";
		close;
	}
	return;
}

// 顯示轉職選單
function	Job_Menu	{
	// getarg(0) 為儲存所有可選轉職職業的 .@job_opt 陣列.
	function Confirm_Change;
	while(true) {
		.@opt_cnt =  getarraysize(getarg(0));
		if( .@opt_cnt <= 0 ) {
			mes "已沒有可轉職的職業.";
			close;
		}

		.@selected = 0; // 僅提供單一職業, 無需選擇
		if (.@opt_cnt > 1) {
			// 提供多個職業選項. 選擇一個並存入 .@class
			// 之後確認 .@class
			mes "請選擇職業.";
			.@menu$ = "";
			for (.@i = 0; .@i < .@opt_cnt; .@i++) {
				if( getelementofarray(getarg(0), .@i) == Job_Novice_High)
					.@jobname$ = "^0055FF轉生^000000";
				else
					.@jobname$ = jobname(getelementofarray(getarg(0), .@i));
				.@menu$ = .@menu$ + " ~ " + .@jobname$ + ":";
			}
			.@menu$ = .@menu$+" ~ ^777777取消^000000";
			.@selected = select(.@menu$) - 1;
			if( .@selected < 0 || .@selected >= .@opt_cnt )
				close;
			next;
			mes .NPCName$;
		}
		.@class = getelementofarray(getarg(0), .@selected);
		if ((.@class == Job_Super_Novice || .@class == Job_Super_Baby) &&
				BaseLevel < .SNovice) {
			// 特殊等級需求, 因為超級初學者與
			// 超級寶寶皆可在其中一個一轉階段被選擇.
			// 這就是為什麼等級需求檢查放在選擇之後而非之前.
			mes "需要基本等級達到 " + .SNovice +
				" 才能轉職為 " + jobname(.@class) + ".";
			return;
		}
		// 確認職業
		Confirm_Change(.@class, .@opt_cnt > 1);
		next;
		mes .NPCName$;
	}
	return;
}


// 執行實際的轉職並關閉對話.
function	Job_Change	{
	.@previous_class = Class;
	.@to_cls = getarg(0);
	next;
	mes .NPCName$;
	mes "你現在成為了 " + jobname(.@to_cls) + "!";
	if (.@to_cls == Job_Novice_High && .LastJob)
		lastJob = Class; // 儲存轉生前的職業
	jobchange .@to_cls;
	if (.@to_cls == Job_Novice_High)
		resetlvl(1);
	else if (.@to_cls == Job_Baby) {
		resetstatus;
		resetskill;
		set SkillPoint,0;
	}
	specialeffect2 EF_ANGEL2;
	specialeffect2 EF_ELECTRIC;
	if (.@previous_class != Class) {
		if (.Platinum)
			callfunc "F_GetPlatinumSkills";
		if (.GetJobEquip)
			Get_Job_Equip();
	}
	close; // 轉職後一律關閉對話
}

function	Confirm_Change	{
	// 玩家確認想要轉職為 .@class
	.@class = getarg(0, -1);
	.@back = getarg(1, false);
	if( .@class < 0 || eaclass(.@class) == -1 ) {
		mes "未知職業錯誤.";
		close;
	}
	mes "你確定要轉職為 ^0055FF"+jobname(.@class)+"^000000 嗎?";
	.@job_option$ = " ~ 轉職為 ^0055FF"+jobname(.@class)+"^000000";
	if( .@class == Job_Novice_High)
		.@job_option$ = " ~ ^0055FF轉生^000000";
	
	if (select(.@job_option$+": ~ ^777777" + 
			((.@back) ?"返回" : "取消") + "^000000") == 1) {
		Job_Change(.@class);
	}
	if (!.@back)
		close; // 按下 "取消"
	return;
}
